64# cos and sin on a grid of [0, 2π), for cos_sin 

65_TRIG_N = 4096 

66_TRIG_STEP = 2 * np.pi / _TRIG_N 

67_TRIG_COS = np.cos(np.arange(_TRIG_N) * _TRIG_STEP) 

68_TRIG_SIN = np.sin(np.arange(_TRIG_N) * _TRIG_STEP)

71@njit(inline="always") 

72def cos_sin(x: float) -> tuple[float, float]: 

73 """Return ``(cos(x), sin(x))``, for use inside numba kernels. 

74 

75 The angle is reduced to [0, 2π), then cos and sin come from the grid 

76 point below it and a Taylor series of the remainder (< 2π/4096), as in 

77 pysanepic. Numba without SVML calls a scalar libm: this is ~5-9 times 

78 faster on Galileo100, ~2 times on Apple M3. Accuracy is ~1e-16, plus 

79 ~|x|·1e-16 from the range reduction. 

80 """ 

81 x = x - 2 * np.pi * np.floor(x * (1 / (2 * np.pi))) 

82 k = min(int(x * (1 / _TRIG_STEP)), _TRIG_N - 1) 

83 d = x - k * _TRIG_STEP 

84 d2 = d * d 

85 cd = 1.0 - d2 * (0.5 - d2 / 24.0) 

86 sd = d * (1.0 - d2 * (1.0 / 6.0 - d2 / 120.0)) 

87 return ( 

88 _TRIG_COS[k] * cd - _TRIG_SIN[k] * sd, 

89 _TRIG_SIN[k] * cd + _TRIG_COS[k] * sd, 

90 )
