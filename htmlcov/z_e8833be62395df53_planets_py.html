1"""Planets: ephemerides seen from the spacecraft and emission. 

2 

3The direction and distance of the planets are computed from the position of 

4the spacecraft (see :class:`.SpacecraftPositionAndVelocity`), so that the 

5parallax is taken into account. The emission is a table of whole-disc 

6thermodynamic temperatures (in K, the temperature of a blackbody with the 

7same brightness, i.e., B(ν, T) = I_ν) with respect to blank sky, i.e., with 

8the CMB monopole occulted by the planet already subtracted, which is what 

9enters a differential TOD. 

10"""

47def _rj_to_thermodynamic(t_rj_k, freq_ghz): 

48 # B(ν, T) = 2kν²/c² T_RJ 

49 x = H_OVER_K_B * np.asarray(freq_ghz) * 1e9 

50 return x / np.log1p(x / np.asarray(t_rj_k))

53# Whole-disc temperatures with respect to blank sky, as (frequencies in GHz, 

54# thermodynamic temperatures in K, i.e., B(ν, T) = I_ν; neither K_CMB nor 

55# K_RJ): 

56# 

57# - 100–857 GHz: Planck HFI, mean over the surveys (Table 4 of Planck 

58# intermediate results LII, A&A 607, A122, 2017, arXiv:1612.07151) 

59# - < 100 GHz for Jupiter and Saturn: Planck LFI (Table 3 of LII; for Saturn 

60# the rings are subtracted, while they are included in the HFI values) 

61# - < 100 GHz for Mars and Uranus: WMAP, Rayleigh-Jeans temperatures (Tables

112def planet_temperature_k(planet: str, freq_ghz) -> npt.NDArray: 

113 """Thermodynamic temperature of `planet` with respect to blank sky, in K 

114 

115 This is the temperature of a blackbody with the same brightness as the 

116 disc of the planet, B(ν, T) = I_ν: it is neither a K_CMB nor a K_RJ 

117 temperature (see :func:`planet_brightness_k_cmb` for the former). The 

118 values in :data:`PLANET_TEMPERATURES_K` are interpolated linearly in 

119 log ν, and kept constant outside the range of the measurements. 

120 """ 

121 freqs, temperatures = PLANET_TEMPERATURES_K[planet] 

122 return np.interp(np.log(freq_ghz), np.log(freqs), temperatures)

125def planet_brightness_k_cmb(planet: str, freqs_ghz, weights=None) -> float: 

126 """Temperature in K_CMB of a source that fills the beam with the 

127 emission of `planet` 

128 

129 This is the ratio between the integrals of w(ν) B(ν, T_planet(ν)) and 

130 w(ν) ∂B/∂T at the temperature of the CMB over the frequencies 

131 `freqs_ghz`, with `weights` w(ν) in power units (uniform if ``None``), 

132 as in PySM 3. With one frequency, the ratio is monochromatic. The 

133 signal of the planet at the centre of the beam is this value times 

134 Ω_planet / Ω_beam. 

135 """ 

136 freqs_hz = 1e9 * np.atleast_1d(np.asarray(freqs_ghz, dtype=np.float64)) 

137 x = H_OVER_K_B * freqs_hz 

138 x_planet = x / planet_temperature_k(planet, freqs_hz / 1e9) 

139 x_cmb = x / T_CMB_K 

140 # Both B(ν, T) and T ∂B/∂T are proportional to ν³ times these factors 

141 radiance = 1 / np.expm1(x_planet) 

142 d_radiance_dt = x_cmb * np.exp(x_cmb) / np.expm1(x_cmb) ** 2 / T_CMB_K 

143 if len(freqs_hz) == 1: 

144 return float(radiance[0] / d_radiance_dt[0]) 

145 

146 w = freqs_hz**3 * (1.0 if weights is None else np.asarray(weights)) 

147 return np.trapezoid(w * radiance, freqs_hz) / np.trapezoid( 

148 w * d_radiance_dt, freqs_hz 

149 )

160def planet_solid_angle_sr(planet: str, distance_km) -> npt.NDArray: 

161 """Solid angle of the disc of `planet` at a distance `distance_km` 

162 

163 ponytail: πr_eq r_pol / R², neglecting the projection of the polar radius 

164 (< 2% for Saturn, less for the others) 

165 """ 

166 r_eq, r_pol = PLANET_RADII_KM[planet] 

167 return np.pi * r_eq * r_pol / np.asarray(distance_km) ** 2

179def planet_ephemeris( 

180 planet: str, 

181 pos_and_vel: SpacecraftPositionAndVelocity, 

182 start_time: astropy.time.Time, 

183 time_span_s: float, 

184) -> PlanetEphemeris: 

185 """Position of `planet` in km and in Ecliptic coordinates on a grid of 

186 times (seconds since `start_time`) covering `time_span_s`, and the orbit 

187 of the spacecraft in `pos_and_vel` 

188 

189 Use :func:`planet_position` to interpolate it. 

190 """ 

191 # ponytail: no light-time correction, the planets move < 0.01° meanwhile 

192 grid_s = np.arange(-1, time_span_s / _EPHEMERIS_STEP_S + 2) * _EPHEMERIS_STEP_S 

193 times = start_time + astropy.time.TimeDelta(grid_s, format="sec") 

194 planet_km = ( 

195 ICRS(get_body_barycentric(planet, times)) 

196 .transform_to(DEFAULT_COORDINATE_SYSTEM) 

197 .cartesian.get_xyz() 

198 .to("km") 

199 .value.T 

200 ) 

201 return PlanetEphemeris( 

202 grid_s=grid_s, 

203 planet_km=planet_km, 

204 spacecraft_offset_s=(start_time - pos_and_vel.start_time).sec, 

205 pos_and_vel=pos_and_vel, 

206 )

209def planet_position( 

210 ephemeris: PlanetEphemeris, times_s: npt.NDArray 

211) -> tuple[npt.NDArray, npt.NDArray]: 

212 """Unit vectors pointing to the planet from the spacecraft and its 

213 distance in km at times `times_s`, interpolating the result of 

214 :func:`planet_ephemeris` 

215 

216 The planet and the spacecraft are interpolated separately at `times_s`: 

217 the grid of the planet extends beyond the orbit of the spacecraft, which 

218 is not extrapolated. 

219 """ 

220 planet_km = np.stack( 

221 [ 

222 np.interp(times_s, ephemeris.grid_s, ephemeris.planet_km[:, i]) 

223 for i in range(3) 

224 ], 

225 axis=-1, 

226 ) 

227 

228 # Same interpolation as SpacecraftPositionAndVelocity.compute_velocities 

229 pos_and_vel = ephemeris.pos_and_vel 

230 tp = np.linspace(0.0, pos_and_vel.time_span_s, pos_and_vel.positions_km.shape[0]) 

231 t = ephemeris.spacecraft_offset_s + np.asarray(times_s) 

232 spacecraft_km = np.stack( 

233 [np.interp(t, tp, pos_and_vel.positions_km[:, i]) for i in range(3)], axis=-1 

234 ) 

235 

236 pos = planet_km - spacecraft_km 

237 distance = np.linalg.norm(pos, axis=1) 

238 return pos / distance[:, None], distance

241def _ang2vec(theta: npt.NDArray, phi: npt.NDArray) -> npt.NDArray: 

242 sin_theta = np.sin(theta) 

243 return np.stack( 

244 [sin_theta * np.cos(phi), sin_theta * np.sin(phi), np.cos(theta)], axis=-1 

245 )

248@njit 

249def _angle_to_planet(pointings, planet_dir, det_idx, sample_idx) -> float: 

250 """Cosine of the angle between a pointing (θ, φ) and the planet""" 

251 theta = pointings[det_idx, sample_idx, 0] 

252 phi = pointings[det_idx, sample_idx, 1] 

253 sin_theta = np.sin(theta) 

254 return ( 

255 sin_theta * np.cos(phi) * planet_dir[sample_idx, 0] 

256 + sin_theta * np.sin(phi) * planet_dir[sample_idx, 1] 

257 + np.cos(theta) * planet_dir[sample_idx, 2] 

258 )

261def _max_detector_offset_rad(obs: Observation) -> float: 

262 """Largest angle between the boresight and the detectors of `obs`""" 

263 max_offset = 0.0 

264 for quat in obs.quat: 

265 z_vectors = np.empty((quat.quats.shape[0], 3)) 

266 all_rotate_z_vectors(z_vectors, quat.quats) 

267 max_offset = max(max_offset, np.arccos(np.clip(z_vectors[:, 2], -1, 1)).max()) 

268 return max_offset

271def _segments_close_to_planet( 

272 obs: Observation, 

273 ephemeris: PlanetEphemeris, 

274 max_distance_rad: float, 

275) -> list[tuple[int, int]]: 

276 """Ranges of samples ``[start, stop)`` in which some detector of `obs` 

277 can be closer than `max_distance_rad` to the planet 

278 

279 The boresight is sampled every _PREFILTER_STEP_S seconds: a segment 

280 between two consecutive points is kept if its first point is closer to the 

281 planet than `max_distance_rad` plus the largest offset of the detectors 

282 and the largest motion of the boresight between two points. 

283 """ 

284 step = max(1, round(_PREFILTER_STEP_S * obs.sampling_rate_hz)) 

285 num_of_points = (obs.n_samples_process - 1) // step + 1 

286 if num_of_points < 2: 

287 return [(0, obs.n_samples_process)] 

288 

289 bore, _ = obs.pointing_provider.get_pointings( 

290 detector_quat=RotQuaternion(), 

291 start_time=obs.start_time_process, 

292 start_time_simulation=obs.start_time_simulation, 

293 sampling_rate_hz=obs.sampling_rate_hz / step, 

294 nsamples=num_of_points, 

295 compute_hwp_angle=False, 

296 ) 

297 bore_vec = _ang2vec(bore[:, 0], bore[:, 1]) 

298 max_motion = np.arccos( 

299 np.clip(np.sum(bore_vec[1:] * bore_vec[:-1], axis=1), -1, 1) 

300 ).max() 

301 

302 times_s = np.arange(num_of_points) * step / obs.sampling_rate_hz 

303 planet_vec, _ = planet_position(ephemeris, times_s) 

304 distance = np.arccos(np.clip(np.sum(bore_vec * planet_vec, axis=1), -1, 1)) 

305 close = distance < max_distance_rad + _max_detector_offset_rad(obs) + max_motion 

306 

307 # Point k covers samples [k * step, (k + 1) * step); join consecutive points 

308 edges = np.flatnonzero(np.diff(np.concatenate([[0], close.astype(int), [0]]))) 

309 return [ 

310 (start * step, min(stop * step, obs.n_samples_process)) 

311 for start, stop in zip(edges[::2], edges[1::2]) 

312 ]

324def iter_planet_segments( 

325 obs: Observation, 

326 pos_and_vel: SpacecraftPositionAndVelocity, 

327 planet: str, 

328 max_distance_rad: float, 

329) -> Iterator[tuple[int, int, npt.NDArray, npt.NDArray, npt.NDArray]]: 

330 """Segments of `obs` in which some detector can be closer than 

331 `max_distance_rad` to `planet` 

332 

333 Yield ``(start, stop, pointings, planet_dir, distance_km)`` for the 

334 samples ``[start, stop)``: the Ecliptic pointings of all the detectors 

335 ``(n_detectors, n, 3)``, the unit vectors pointing to the planet 

336 ``(n, 3)`` and its distance from the spacecraft ``(n,)``. The pointings 

337 are computed only in these segments, found sampling the boresight every 

338 second. 

339 """ 

340 if obs.n_samples_process == 0 or obs.n_detectors == 0: 

341 return 

342 

343 ephemeris = planet_ephemeris( 

344 planet, 

345 pos_and_vel, 

346 obs.start_time_process, 

347 obs.n_samples_process / obs.sampling_rate_hz, 

348 ) 

349 for start, stop in _segments_close_to_planet(obs, ephemeris, max_distance_rad): 

350 pointings = np.empty((obs.n_detectors, stop - start, 3)) 

351 for det_idx in range(obs.n_detectors): 

352 obs.pointing_provider.get_pointings( 

353 detector_quat=obs.quat[det_idx], 

354 start_time=obs.start_time_process 

355 + astropy.time.TimeDelta(start / obs.sampling_rate_hz, format="sec"), 

356 start_time_simulation=obs.start_time_simulation, 

357 sampling_rate_hz=obs.sampling_rate_hz, 

358 nsamples=stop - start, 

359 pointing_buffer=pointings[det_idx], 

360 compute_hwp_angle=False, 

361 ) 

362 times_s = np.arange(start, stop) / obs.sampling_rate_hz 

363 planet_dir, distance_km = planet_position(ephemeris, times_s) 

364 yield start, stop, pointings, planet_dir, distance_km

371def planet_beam_parameters( 

372 obs: Observation, planet: str, bandpass_integration: bool = False 

373) -> tuple[npt.NDArray, npt.NDArray]: 

374 """Return σ of the Gaussian beams and the signal at their centre per 

375 steradian of `planet` (in K_CMB/sr) for each detector of `obs` 

376 

377 With `bandpass_integration`, the emission is integrated over the 

378 bandpass of each detector (``obs.band``, or a top-hat band from 

379 ``bandcenter_ghz`` and ``bandwidth_ghz`` if missing); otherwise, it is 

380 computed at ``bandcenter_ghz``, as in :class:`.SkyGenerationParams`. 

381 """ 

382 fwhm_rad = np.deg2rad(np.asarray(obs.fwhm_arcmin) / 60.0) 

383 sigma_rad = fwhm_rad / np.sqrt(8 * np.log(2)) 

384 brightness = np.empty(obs.n_detectors) 

385 for det_idx in range(obs.n_detectors): 

386 if bandpass_integration: 

387 band = _detector_bandpass(obs, det_idx) 

388 brightness[det_idx] = planet_brightness_k_cmb( 

389 planet, band.freqs_ghz, band.weights 

390 ) 

391 else: 

392 brightness[det_idx] = planet_brightness_k_cmb( 

393 planet, obs.bandcenter_ghz[det_idx] 

394 ) 

395 return sigma_rad, brightness / (2 * np.pi * sigma_rad**2)

398def _k_cmb_to_tod_units( 

399 obs: Observation, component: str, bandpass_integration: bool 

400) -> npt.NDArray: 

401 # Factors converting K_CMB into the units of the TOD `component`, for 

402 # each detector (K_CMB if the component is not in obs.tod_list) 

403 units = next((t.units for t in obs.tod_list if t.name == component), Units.K_CMB) 

404 if units == Units.K_CMB: 

405 return np.ones(obs.n_detectors) 

406 return np.array( 

407 [ 

408 _get_cmb_unit_conversion( 

409 units, 

410 Units.K_CMB, 

411 freq_ghz=None if bandpass_integration else obs.bandcenter_ghz[i], 

412 bandpass=_detector_bandpass(obs, i) if bandpass_integration else None, 

413 band_integration=bandpass_integration, 

414 ) 

415 for i in range(obs.n_detectors) 

416 ] 

417 )

420@njit 

421def add_planet_signal( 

422 tod, 

423 pointings, 

424 planet_dir, 

425 planet_solid_angle, 

426 peak_per_sr, 

427 sigma_rad, 

428 max_angle_rad, 

429) -> None: 

430 """Add the signal of a planet seen through Gaussian beams to `tod` 

431 

432 `tod` (``(n_detectors, n_samples)``) is modified in place; `pointings` 

433 (``(n_detectors, n_samples, ≥2)``) contains θ and φ, `planet_dir` 

434 (``(n_samples, 3)``) the unit vectors pointing to the planet, 

435 `planet_solid_angle` (``(n_samples,)``) its solid angle, `peak_per_sr` 

436 (``(n_detectors,)``) the signal at the centre of the beam per steradian 

437 of planet, and `sigma_rad` (``(n_detectors,)``) the width of the beams, 

438 truncated at `max_angle_rad` (``(n_detectors,)``). 

439 """ 

440 for det_idx in range(tod.shape[0]): 

441 cos_max = np.cos(max_angle_rad[det_idx]) 

442 inv_two_sigma2 = 0.5 / sigma_rad[det_idx] ** 2 

443 for sample_idx in range(tod.shape[1]): 

444 cos_angle = _angle_to_planet(pointings, planet_dir, det_idx, sample_idx) 

445 if cos_angle > cos_max: 

446 angle = np.arccos(min(cos_angle, 1.0)) 

447 tod[det_idx, sample_idx] += ( 

448 peak_per_sr[det_idx] 

449 * planet_solid_angle[sample_idx] 

450 * np.exp(-(angle**2) * inv_two_sigma2) 

451 )

454def add_planets_to_observations( 

455 observations: Observation | list[Observation], 

456 pos_and_vel: SpacecraftPositionAndVelocity | None, 

457 planets: str | list[str] | tuple[str, ...] = PLANETS, 

458 component: str = "tod", 

459 bandpass_integration: bool = False, 

460) -> None: 

461 """Add the signal of the planets to the TODs 

462 

463 The planets are unpolarized point sources with the temperatures in 

464 :data:`PLANET_TEMPERATURES_K`, seen through circular Gaussian beams 

465 (``fwhm_arcmin``), truncated at 3 FWHM. Their emission is computed at 

466 ``bandcenter_ghz`` or, with `bandpass_integration`, integrated over the 

467 bandpass of each detector (see :func:`planet_beam_parameters`). The 

468 signal is converted from K_CMB into the units of `component` declared 

469 in ``obs.tod_list``. Their direction and 

470 solid angle are computed from the position of the spacecraft in 

471 `pos_and_vel`. As for :func:`.flag_planets_in_observations`, the pointings 

472 are computed only close to the planets, and the times of the 

473 observations must be ``astropy.time.Time`` objects. 

474 

475 `pos_and_vel` may be None only when `observations` is empty. 

476 

477 ponytail: Gaussian beams; the radial profile of the beam would give 

478 realistic sidelobes. 

479 """ 

480 obs_list = [observations] if isinstance(observations, Observation) else observations 

481 planets = [planets] if isinstance(planets, str) else list(planets) 

482 check_observations_for_planets(obs_list) 

483 _check_detector_fields(obs_list, ["fwhm_arcmin"], component) 

484 

485 if pos_and_vel is None: 

486 if obs_list: 

487 raise ValueError("Planets need pos_and_vel when observations are not empty") 

488 return 

489 

490 for obs in obs_list: 

491 tod = getattr(obs, component) 

492 fwhm_rad = np.deg2rad(np.asarray(obs.fwhm_arcmin) / 60.0) 

493 max_angle_rad = _BEAM_TRUNCATION_FWHM * fwhm_rad 

494 to_tod_units = _k_cmb_to_tod_units(obs, component, bandpass_integration) 

495 for planet in planets: 

496 sigma_rad, peak_per_sr = planet_beam_parameters( 

497 obs, planet, bandpass_integration 

498 )
