1"""Detector and time splits of the map-makers 

2 

3Each split name is bound to the function that builds its mask, so that the 

4names accepted by :func:`.check_valid_splits` and those that the map-makers 

5can build are the same by construction. The map-makers validate the splits 

6every time they build a mask, also when they are called without a 

7:class:`.Simulation`. 

8"""

10from collections.abc import Callable 

11 

12import astropy.time 

13import numpy as np 

14import numpy.typing as npt

16from litebird_sim.observations import Observation

18#: Length of a year and of a survey (half a year) in the time splits, in seconds 

19T_YEAR_S = 365 * 24 * 3600 

20T_SURVEY_S = T_YEAR_S / 2 

21 

22MAX_YEARS = 3 

23MAX_SURVEYS = 6 

24 

25#: Wafers of the LiteBIRD focal plane, accepted as ``"wafer<name>"`` 

26LITEBIRD_WAFERS = ( 

27 [f"L{i:02d}" for i in range(8)] 

28 + [f"M{i:02d}" for i in range(7)] 

29 + [f"H{i:02d}" for i in range(3)] 

30)

33def _get_initial_time(obs: Observation) -> float: 

34 if isinstance(obs.start_time_observation, astropy.time.Time): 

35 return obs.start_time_observation.cxcsec 

36 return obs.start_time_observation

39def _get_end_time(obs: Observation) -> float: 

40 if isinstance(obs.end_time_observation, astropy.time.Time): 

41 return obs.end_time_observation.cxcsec 

42 return obs.end_time_observation

45def _full(obs: Observation) -> npt.NDArray: 

46 # A read-only view needs no memory: a real array would take one byte per 

47 # sample for *all* the observations at once (~7 GB per process for 3 years 

48 # of one 75 Hz detector) 

49 return np.broadcast_to(np.True_, obs.n_samples_process)

52def _samples(selection: Callable[[int], slice]) -> Callable: 

53 def build(obs: Observation) -> npt.NDArray: 

54 mask = np.zeros(obs.n_samples_process, dtype=bool) 

55 mask[selection(obs.n_samples_process)] = True 

56 return mask 

57 

58 return build

61def _period(index: int, length_s: float) -> Callable: 

62 def build(obs: Observation) -> npt.NDArray: 

63 times = obs.get_times() - _get_initial_time(obs) 

64 return (times >= (index - 1) * length_s) & (times < index * length_s) 

65 

66 return build 

67

69#: Time split name → function returning the mask of one observation 

70TIME_SPLITS: dict[str, Callable[[Observation], npt.NDArray]] = { 

71 "full": _full, 

72 "odd": _samples(lambda n: slice(0, None, 2)), 

73 "even": _samples(lambda n: slice(1, None, 2)), 

74 "first_half": _samples(lambda n: slice(0, n // 2)), 

75 "second_half": _samples(lambda n: slice(n // 2, None)), 

76 **{f"year{i}": _period(i, T_YEAR_S) for i in range(1, MAX_YEARS + 1)}, 

77 **{f"survey{i}": _period(i, T_SURVEY_S) for i in range(1, MAX_SURVEYS + 1)}, 

78} 

79 

80#: Valid detector split names 

81DETECTOR_SPLITS = ["full"] + [f"wafer{w}" for w in LITEBIRD_WAFERS] 

82

84def _validate_time_split(time_split: str, obs_list: list[Observation]) -> None: 

85 if time_split not in TIME_SPLITS: 

86 raise ValueError( 

87 f"Unknown time split '{time_split}', valid ones are {list(TIME_SPLITS)}" 

88 ) 

89 for prefix, length_s in (("year", T_YEAR_S), ("survey", T_SURVEY_S)): 

90 if not time_split.startswith(prefix): 

91 continue 

92 index = int(time_split.removeprefix(prefix)) 

93 for cur_obs in obs_list: 

94 duration = round(_get_end_time(cur_obs) - _get_initial_time(cur_obs), 0) 

95 if index > duration // length_s: 

96 raise AssertionError( 

97 f"Time split '{time_split}' not possible for an observation " 

98 f"lasting {round(duration / T_YEAR_S, 2)} years" 

99 )

102def _validate_detector_split(detector_split: str, obs_list: list[Observation]) -> None: 

103 if detector_split not in DETECTOR_SPLITS: 

104 raise ValueError( 

105 f"Unknown detector split '{detector_split}', valid ones are " 

106 f"{DETECTOR_SPLITS}" 

107 ) 

108 if detector_split == "full": 

109 return 

110 wafer = detector_split.removeprefix("wafer") 

111 for cur_obs in obs_list: 

112 if cur_obs.wafer is None or wafer not in cur_obs.wafer: 

113 raise AssertionError( 

114 f"The requested wafer '{wafer}' is not in the observation, " 

115 f"which has the wafers {cur_obs.wafer}" 

116 )

119def _build_mask_time_split( 

120 time_split: str, obs_list: list[Observation] 

121) -> list[npt.NDArray]: 

122 """One Boolean mask ``(n_samples,)`` per observation, True for the samples 

123 in `time_split`. Raise if the split is unknown or not possible.""" 

124 _validate_time_split(time_split, obs_list) 

125 build = TIME_SPLITS[time_split] 

126 return [build(cur_obs) for cur_obs in obs_list]

129def _build_mask_detector_split( 

130 detector_split: str, obs_list: list[Observation] 

131) -> list[npt.NDArray]: 

132 """One Boolean mask ``(n_detectors,)`` per observation, True for the 

133 detectors in `detector_split`. Raise if the split is unknown or not 

134 possible.""" 

135 _validate_detector_split(detector_split, obs_list) 

136 if detector_split == "full": 

137 return [np.ones(cur_obs.n_detectors, dtype=bool) for cur_obs in obs_list] 

138 wafer = detector_split.removeprefix("wafer") 

139 return [np.asarray(cur_obs.wafer) == wafer for cur_obs in obs_list]

142def check_valid_splits( 

143 observations: Observation | list[Observation], 

144 detector_splits: str | list[str] = "full", 

145 time_splits: str | list[str] = "full", 

146): 

147 """Check if the splits are valid 

148 

149 For each observation in the list, check if the detector and time splits 

150 are valid. 

151 In particular, the compatibility between the detectors in each observation 

152 and the desired split in detector domain is checked. On the other hand, this 

153 assess whether the desired time split fits inside the duration of the 

154 observation (when this applies). 

155 If the splits are not compatible with the input data, an error is raised: 

156 ``ValueError`` for an unknown split, ``AssertionError`` for a split that 

157 the observations cannot satisfy. 

158 

159 Args: 

160 observations (list of :class:`Observations`): observations to be mapped. 

161 They are required to have the following attributes as arrays 

162 

163 * `pointings`: the pointing information (in radians) for each tod 

164 sample. It must be a tensor with shape ``(N_d, N_t, 3)``, 

165 with ``N_d`` number of detectors and ``N_t`` number of 

166 samples in the TOD. 

167 * any attribute listed in `components` (by default, `tod`) and 

168 containing the TOD(s) to be binned together. 

169 

170 If the observations are distributed over some communicator(s), they 

171 must share the same group processes. 

172 If pointings and psi are not included in the observations, they can 

173 be provided through an array (or a list of arrays) of dimension 

174 (Ndetectors x Nsamples x 3), containing theta, phi and psi 

175 detector_splits (str | list[str], optional): detector-domain splits 

176 used to produce maps. 

177 

178 * "full": every detector in the observation will be used; 

179 * "waferXXX": the mapmaking will be performed on the intersection 

180 of the detectors specified in the input and the detectors specified 

181 in the detector_split. 

182 The wafer must be specified in the format "waferXXX". The valid values 

183 for "XXX" are all the 3-digits strings corresponding to the wafers 

184 in the LITEBIRD focal plane (e.g. L00, M01, H02). 

185 

186 time_splits (str | list[str], optional): time-domain splits 

187 used to produce maps. This defaults to "full" indicating that every 

188 sample in the observation will be used. In addition, the user can specify 

189 a string, or a list of strings, to indicate a subsample of the observation 

190 to be used: 

191 

192 * "full": every sample in the observation will be used; 

193 * "first_half" and/or "second_half": the first and/or second half of the 

194 observation will be used; 

195 * "odd" and/or "even": the odd and/or even samples in the observation 

196 will be used; 

197 * "yearX": the samples in the observation will be 

198 used according to the year they belong to (relative to the 

199 starting time). The valid values for "X" are ["1", "2", "3"]. 

200 * "surveyX": the samples in the observation will be used according 

201 to the requested survey. In this context, a survey is taken to 

202 be complete in 6 months, thus the valid values for "X" are 

203 ["1", "2", "3", "4", "5", "6"]. 

204 

205 """ 

206 if isinstance(observations, Observation): 

207 observations = [observations] 

208 if isinstance(detector_splits, str): 

209 detector_splits = [detector_splits] 

210 if isinstance(time_splits, str): 

211 time_splits = [time_splits] 

212 

213 for detector_split in detector_splits: 

214 _validate_detector_split(detector_split, observations) 

215 for time_split in time_splits: 

216 _validate_time_split(time_split, observations)
